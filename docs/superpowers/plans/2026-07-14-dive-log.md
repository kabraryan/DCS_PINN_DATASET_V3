# Dive Log Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a fourth tab to DecoStress that saves scenarios built in the Risk explorer and reviews them via a comparison table, an overlay chart, and a per-dive driver breakdown.

**Architecture:** Pure addition to the existing single-file app. A store layer over `localStorage` persists only the polyline; every displayed number is re-derived at render by calling the existing `predict()` / `simulatePolyline()`. No backend, no build, no new dependencies.

**Tech Stack:** Vanilla JS + inline CSS in `decostress_app/index.html`. Python guards in `tests/test_web_model_sync.py` (pytest). Behavioural tests via jsdom.

## Global Constraints

- **Store inputs, not outputs.** `localStorage` holds `{id, name, savedAt, poly}` and nothing else. Derived fields (`pct`, `score`, `requiredAscent`, `deficit`, `stress`) MUST NOT be persisted. Refitting the model must automatically re-score saved dives.
- **A voided dive has NO model rank.** The table renders the literal string `VOID`, never a number. Sorting by rank groups voided dives as a block at the end; they are never interleaved with valid dives.
- **Never emit a probability.** The rank is a percentile within the 1,948 real NMRC 99-02 dives. No `P(DCS)` token may reach the user.
- **Storage key is versioned:** `decostress.scenarios.v1`. Unknown versions are ignored, not crashed on.
- **Corrupt storage must not brick the page.** Discard, log to console, start empty.
- Python: `/opt/miniconda3/bin/python3`. jsdom lives at `$CLAUDE_JOB_DIR/tmp/node_modules/jsdom`.

/* ================================================================
   DIVE LOG — store layer.

   Persists ONLY the polyline. Rank, obligation, loading and violation are
   re-derived at render by predict()/simulatePolyline(). Caching them would
   freeze them: refit the model and every saved dive would keep serving a
   stale number, with nothing to catch it. That is Correction 10 — a
   hand-copied value diverging from its source — and the whole repo is
   guarded against it. The log does not get to reintroduce it.
   ================================================================ */
const LOG_KEY='decostress.scenarios.v1';

function logLoad(){
  try{
    const raw=localStorage.getItem(LOG_KEY);
    if(!raw) return [];
    const list=JSON.parse(raw);
    if(!Array.isArray(list)) return [];
    // Keep only well-formed records. A corrupt entry must not brick the page.
    return list.filter(s=>s && typeof s.id==='string' && Array.isArray(s.poly) && s.poly.length>=2);
  }catch(e){
    console.warn('dive log: unreadable storage, starting empty', e);
    return [];
  }
}
function logSave(list){
  try{ localStorage.setItem(LOG_KEY, JSON.stringify(list)); }
  catch(e){ console.warn('dive log: could not persist', e); }
}
function logAdd(name, poly){
  if(!Array.isArray(poly) || poly.length<2) throw new Error('a scenario needs at least two waypoints');
  const s={
    id:'s_'+Date.now()+'_'+Math.random().toString(36).slice(2,6),
    name:String(name||'').trim() || defaultName(poly),
    savedAt:Date.now(),
    poly:poly.map(p=>({t:+p.t, d:+p.d}))   // inputs only — never the derived scores
  };
  const list=logLoad(); list.push(s); logSave(list);
  return s;
}
function logRemove(id){ logSave(logLoad().filter(s=>s.id!==id)); }
function defaultName(poly){
  const sc=diveScalars(poly);
  if(!sc) return 'dive';
  return Math.round(sc.maxDepthM)+' m / '+Math.round(sc.bottom)+' min';
}
function logExport(){ return JSON.stringify(logLoad(), null, 2); }
function logImport(json){
  let incoming;
  try{ incoming=JSON.parse(json); }catch(e){ throw new Error('that file is not valid JSON'); }
  if(!Array.isArray(incoming)) throw new Error('expected a list of scenarios');
  const list=logLoad(), have=new Set(list.map(s=>s.id));
  let added=0, skipped=0;
  for(const s of incoming){
    if(!s || typeof s.id!=='string' || !Array.isArray(s.poly) || s.poly.length<2){ skipped++; continue; }
    if(have.has(s.id)){ skipped++; continue; }
    list.push({id:s.id, name:String(s.name||'dive'), savedAt:+s.savedAt||Date.now(),
               poly:s.poly.map(p=>({t:+p.t, d:+p.d}))});
    have.add(s.id); added++;
  }
  logSave(list);
  return {added, skipped};
}
```

/* ---- dive log: render ---------------------------------------------------
   Every number here is derived NOW from the stored polyline. Nothing is read
   back from storage except the profile itself.                              */
let logSortKey='savedAt', logSortDir=-1;

function logRows(){
  return logLoad().map(s=>{
    const r=predict(s.poly);
    const stress=simulatePolyline(s.poly).st;
    // A voided dive HAS NO RANK. Not a low one, not a high one -- none.
    // The model never saw a dive that skipped its obligation, and its positive
    // ascent coefficient would call that dive safer. A number in this column
    // would read as comparable to the honest ones. It must not exist.
    const rank = (r && !r.violated) ? r.pct : null;
    return {s, r, stress, rank};
  }).filter(x=>x.r);
}
function sortLog(key){
  if(logSortKey===key) logSortDir*=-1; else { logSortKey=key; logSortDir=1; }
  renderLog();
}
function logSorted(){
  const rows=logRows();
  const val={
    name:x=>x.s.name.toLowerCase(), depth:x=>x.r.depthFsw, bottom:x=>x.r.bottom,
    ascent:x=>x.r.actualAscent, req:x=>x.r.requiredAscent, stress:x=>x.stress,
    savedAt:x=>x.s.savedAt,
  };
  if(logSortKey==='rank'){
    // Voided dives are not comparable. Sort the valid ones, then append the
    // voided block -- never interleaved.
    const valid=rows.filter(x=>x.rank!=null).sort((a,b)=>(a.rank-b.rank)*logSortDir);
    const void_=rows.filter(x=>x.rank==null).sort((a,b)=>b.stress-a.stress);
    return valid.concat(void_);
  }
  const f=val[logSortKey]||val.savedAt;
  return rows.sort((a,b)=>{ const A=f(a),B=f(b); return (A<B?-1:A>B?1:0)*logSortDir; });
}
function ord2(n){ const s=['th','st','nd','rd'],v=n%100; return n+(s[(v-20)%10]||s[v]||s[0]); }

function renderLog(){
  const rows=logSorted();
  $('logCount').textContent = rows.length+(rows.length===1?' scenario':' scenarios');
  $('logEmpty').hidden = rows.length>0;
  $('logHas').hidden   = rows.length===0;
  if(!rows.length){ $('logTbody').innerHTML=''; $('logDrivers').innerHTML=''; return; }

  $('logTbody').innerHTML = rows.map(x=>{
    const r=x.r, v=r.violated;
    return '<tr class="'+(v?'void':'')+'">'+
      '<td>'+esc(x.s.name)+'</td>'+
      '<td>'+Math.round(r.depthFsw/M_TO_FSW)+' m</td>'+
      '<td>'+r.bottom.toFixed(0)+' min</td>'+
      '<td>'+r.actualAscent.toFixed(1)+' min</td>'+
      '<td>'+r.requiredAscent.toFixed(1)+' min</td>'+
      '<td>'+(x.rank==null ? '<span class="rankvoid">VOID</span>' : ord2(Math.round(x.rank*100)))+'</td>'+
      '<td>'+x.stress.toFixed(2)+'×</td>'+
      '<td>'+(v ? '<span class="con">DECO VIOLATION</span>' : '<span class="pro">OK</span>')+'</td>'+
      '<td><button class="del" data-del="'+x.s.id+'" title="Remove">✕</button></td>'+
    '</tr>';
  }).join('');
  $('logTbody').querySelectorAll('[data-del]').forEach(b=>
    b.onclick=()=>{ logRemove(b.dataset.del); renderLog(); });

  $('logDrivers').innerHTML = rows.map(x=>{
    const r=x.r;
    if(r.violated){
      return '<div class="logdrv void"><h4>'+esc(x.s.name)+' — model void</h4>'+
        '<p>This dive skipped <b>'+r.deficit.toFixed(1)+' min</b> of its '+
        r.requiredAscent.toFixed(1)+'-minute decompression obligation, and Bühlmann puts it at <b>'+
        x.stress.toFixed(2)+'×</b> the surfacing limit. The model <b>cannot score this dive</b>: every diver '+
        'in its training set followed their schedule, so it has never seen one like this — and its positive '+
        'ascent coefficient would call the short ascent <i>safer</i>. No rank is shown, because none is honest.</p></div>';
    }
    const bars=r.drivers.filter(d=>Math.abs(d.z)>0.001).map(d=>
      (d.z>=0?'▲ ':'▼ ')+(d.name==='ascent time'?'deco obligation':d.name)+
      ': <b>'+(d.z>=0?'+':'−')+Math.abs(d.z).toFixed(3)+'</b>').join(' &nbsp;·&nbsp; ');
    return '<div class="logdrv"><h4>'+esc(x.s.name)+' — '+ord2(Math.round(x.rank*100))+' percentile</h4>'+
      '<p>'+bars+'<br><span style="color:var(--muted)">Standardised log-odds. The obligation dominates; '+
      'max depth on its own barely registers.</span></p></div>';
  }).join('');

  drawLogChart(rows);
}
function esc(s){ return String(s).replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c])); }

const LOG_COLOURS=['#1fc7d4','#2fbf9e','#f2a63a','#9b8cff','#5ec2f2','#d98cc4'];
function drawLogChart(rows){
  const cv=$('logChart'); if(!cv) return;
  const ctx=cv.getContext('2d'); if(!ctx) return;
  const W=cv.width,H=cv.height,PADL=44,PADR=14,PADT=16,PADB=30;
  ctx.clearRect(0,0,W,H);
  const maxT=Math.max(10,...rows.map(x=>x.s.poly[x.s.poly.length-1].t));
  const maxD=Math.max(10,...rows.map(x=>Math.max(...x.s.poly.map(p=>p.d))));
  const X=t=>PADL+(t/maxT)*(W-PADL-PADR);
  const Y=d=>PADT+(d/maxD)*(H-PADT-PADB);

  ctx.strokeStyle='#1c4257'; ctx.fillStyle='#6b93a5';
  ctx.font='11px IBM Plex Mono, monospace'; ctx.lineWidth=1;
  for(let i=0;i<=4;i++){ const d=maxD*i/4;
    ctx.beginPath(); ctx.moveTo(PADL,Y(d)); ctx.lineTo(W-PADR,Y(d)); ctx.stroke();
    ctx.textAlign='right'; ctx.fillText(Math.round(d)+'m',PADL-6,Y(d)+4); }
  ctx.textAlign='center'; ctx.fillText('minutes',(PADL+W-PADR)/2,H-6);

  rows.forEach((x,i)=>{
    const col = x.r.violated ? '#e2574c' : LOG_COLOURS[i%LOG_COLOURS.length];
    ctx.strokeStyle=col; ctx.lineWidth=2;
    if(x.r.violated) ctx.setLineDash([5,3]); else ctx.setLineDash([]);
    ctx.beginPath();
    x.s.poly.forEach((p,j)=> j?ctx.lineTo(X(p.t),Y(p.d)):ctx.moveTo(X(p.t),Y(p.d)));
    ctx.stroke(); ctx.setLineDash([]);
    ctx.fillStyle=col; ctx.textAlign='left';
    ctx.fillText('■ '+x.s.name, PADL+8, PADT+12+i*14);
  });
}

def test_dive_log_persists_inputs_not_derived_scores():
    """The log must store the polyline and re-derive everything else.

    Caching rank/obligation/loading would freeze them: refit the model and every
    saved dive keeps serving a stale number with nothing to catch it. That is the
    Correction 10 failure -- a value hand-copied away from its source -- and the
    rest of this repo is guarded against it. The log does not get an exemption.
    """
    text = _html()
    m = re.search(r"function logAdd\([^)]*\)\s*\{(.*?)\n\}", text, re.S)
    assert m, "logAdd not found"
    body = m.group(1)
    assert "poly:" in body, "logAdd must persist the polyline"
    for derived in ("pct", "score", "requiredAscent", "deficit", "stress", "violated"):
        assert f"{derived}:" not in body, (
            f"logAdd persists the derived field {derived!r}. Store inputs, not outputs -- "
            "a cached score cannot re-derive itself when the model is refitted."
        )


def test_dive_log_never_shows_a_rank_for_a_voided_dive():
    """A sortable table is exactly where the reassuring-number bug would creep back.

    The explorer voids its dial on a decompression violation because the fitted
    model's ascent coefficient is positive -- on a dive that skipped its deco it
    would call a faster ascent safer. If the log printed a tidy percentile next to
    the honest dives, that number would read as comparable and voiding the dial
    would have been pointless.
    """
    text = _html()
    m = re.search(r"function logRows\(\)\s*\{(.*?)\n\}", text, re.S)
    assert m, "logRows not found"
    assert re.search(r"rank\s*=\s*\(?\s*r\s*&&\s*!r\.violated", m.group(1)), (
        "logRows must yield rank=null for a violated dive, so the table cannot print one"
    )
    assert "VOID" in text, "the table must render the literal string VOID for a voided dive"
    # sorting must not interleave voided dives among valid ones
    m2 = re.search(r"if\(logSortKey==='rank'\)\{(.*?)\n\s{2}\}", text, re.S)
    assert m2, "rank sorting branch not found"
    assert "concat" in m2.group(1), (
        "sorting by rank must append the voided block after the valid dives, "
        "never interleave them as though their rank were comparable"
    )


def test_dive_log_storage_key_is_versioned():
    text = _html()
    assert "decostress.scenarios.v1" in text, "the storage key must be versioned"
```
